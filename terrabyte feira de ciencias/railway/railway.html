const PASSENGER_SPEED_REFERENCE_KMH = 1.5; const PASSENGER_SPEED_MAX_KMH = 2.0; const PASSENGER_SPEED_REFERENCE_MS = PASSENGER_SPEED_REFERENCE_KMH / 3.6; // ============================================================ // ELEMENTOS // ============================================================ const railwaySvg = document.getElementById("railwaySimulation"); const passengerPath = document.getElementById("passengerRailwayPath"); const passengerTrain = document.getElementById("passengerTrain"); const startButton = document.getElementById("startButton"); const pauseButton = document.getElementById("pauseButton"); const resetButton = document.getElementById("resetButton"); const speedDisplay = document.getElementById("speedDisplay"); // ============================================================ // ESTADO // ============================================================ let running = false; let distanceMeters = 0; let previousTimestamp = null; // ============================================================ // IMPORTANTE SOBRE ESCALA // ============================================================ // // O comprimento físico real da ferrovia ainda precisa ser // medido no protótipo. // // Portanto, esta variável deve ser preenchida quando a // medição real estiver disponível. // // NÃO ALTERAR A VELOCIDADE DO TREM PARA COMPENSAR A ESCALA. // // A velocidade física permanece 1,5 km/h. // A escala do circuito deve ser calibrada separadamente. // const PASSENGER_TRACK_LENGTH_METERS = null; // ============================================================ // CONVERSÃO DE DISTÂNCIA FÍSICA PARA SVG // ============================================================ function getPassengerProgress() { /* * Quando o comprimento físico real do circuito for * informado, a posição será calculada com base nele. * * Enquanto isso, usamos uma representação proporcional * do percurso para permitir o funcionamento da animação. */ if (PASSENGER_TRACK_LENGTH_METERS === null) { return 0; } const progress = (distanceMeters % PASSENGER_TRACK_LENGTH_METERS) / PASSENGER_TRACK_LENGTH_METERS; return progress; } // ============================================================ // POSIÇÃO DO TREM // ============================================================ function updatePassengerTrain(progress) { const svgPathLength = passengerPath.getTotalLength(); const currentLength = progress * svgPathLength; const lookAheadDistance = 2; const currentPoint = passengerPath.getPointAtLength(currentLength); const nextPoint = passengerPath.getPointAtLength( (currentLength + lookAheadDistance) % svgPathLength ); const angle = Math.atan2( nextPoint.y - currentPoint.y, nextPoint.x - currentPoint.x ) * 180 / Math.PI; passengerTrain.setAttribute( "transform", `translate(${currentPoint.x} ${currentPoint.y}) rotate(${angle})` ); } // ============================================================ // LOOP DA SIMULAÇÃO // ============================================================ function animationLoop(timestamp) { if (!running) { previousTimestamp = null; return; } if (previousTimestamp === null) { previousTimestamp = timestamp; } const deltaTime = (timestamp - previousTimestamp) / 1000; previousTimestamp = timestamp; /* * MOVIMENTO FÍSICO: * * distância = velocidade × tempo * * A velocidade permanece 1,5 km/h. */ distanceMeters += PASSENGER_SPEED_REFERENCE_MS * deltaTime; /* * Enquanto o comprimento real da pista ainda não estiver * medido, não fazemos uma conversão arbitrária para SVG. * * O ponto abaixo será ativado após a medição. */ if (PASSENGER_TRACK_LENGTH_METERS !== null) { const progress = getPassengerProgress(); updatePassengerTrain(progress); } requestAnimationFrame(animationLoop); } // ============================================================ // CONTROLES // ============================================================ startButton.addEventListener("click", () => { if (!running) { running = true; requestAnimationFrame(animationLoop); } }); pauseButton.addEventListener("click", () => { running = false; previousTimestamp = null; }); resetButton.addEventListener("click", () => { running = false; distanceMeters = 0; previousTimestamp = null; /* * Posição inicial do trem. */ updatePassengerTrain(0); }); speedDisplay.textContent = `${PASSENGER_SPEED_REFERENCE_KMH.toFixed(1).replace(".", ",")} km/h`; // ============================================================ // POSIÇÃO INICIAL // ============================================================ updatePassengerTrain(0); ```
